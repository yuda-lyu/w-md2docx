/**
 * 取錯誤訊息文字
 *
 * 底層轉檔器之reject可能為字串或Error物件，統一轉為文字供組合訊息使用。
 *
 * @param {*} err 輸入錯誤資料，可為字串、Error物件或任意資料
 * @returns {String} 回傳錯誤訊息字串
 * @example
 *
 * console.log(toErrText('abc'))
 * // => 'abc'
 *
 * console.log(toErrText(new Error('abc')))
 * // => 'abc'
 *
 */
function toErrText(err) {
    if (isstr(err)) {
        return err
    }
    if (isestr(get(err, 'message'))) {
        return err.message
    }
    let s = ''
    try {
        s = JSON.stringify(err) //undefined/function/symbol 會回 undefined 而非字串
    }
    catch (e) {
        s = ''
    }
    if (!isstr(s)) {
        s = String(err)
    }
    return s
}


/**
 * 重試短暫被鎖定之檔案操作
 *
 * Windows下新建檔案常被防毒或索引器即時掃描而短暫鎖定(EBUSY/EPERM)，遇此類錯誤時延遲後重試以提高穩定性，其餘錯誤直接拋出。
 *
 * @param {Function} fun 輸入欲執行之函數，可回傳Promise
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Integer} [opt.tries=6] 輸入最多嘗試次數整數，預設6
 * @param {Number} [opt.delayMs=400] 輸入每次重試前延遲毫秒數，預設400
 * @returns {Promise} 回傳Promise，resolve回傳fun之結果，reject回傳錯誤
 * @example
 *
 * let r = await retryBusy(() => fs.promises.readFile('./a.txt', 'utf8'), { tries: 3 })
 *
 */
async function retryBusy(fun, opt = {}) {

    //check
    if (!isfun(fun)) {
        return Promise.reject('fun must be a function')
    }

    //tries
    let tries = get(opt, 'tries', null)
    if (!ispint(tries)) {
        tries = 6
    }
    tries = cint(tries)

    //delayMs
    let delayMs = get(opt, 'delayMs', null)
    if (!ispnum(delayMs)) {
        delayMs = 400
    }
    delayMs = cdbl(delayMs)

    let errLast = null
    for (let i = 0; i < tries; i++) {
        try {
            return await fun()
        }
        catch (err) {
            errLast = err
            let code = get(err, 'code', '')
            let busy = (code === 'EBUSY' || code === 'EPERM')
            if (busy && i < tries - 1) {
                await new Promise((resolve) => setTimeout(resolve, delayMs))
                continue
            }
            throw err
        }
    }
    throw errLast
}


//pmChain: docx 轉檔須逐一序列化
//why: 底層係調用本機 Word(win32com)進行轉檔, 同時執行多份會相互干擾(共用 Word 實例、模板檔遭鎖定),
//     故不論同時進來幾個請求, docx 階段一律排隊逐一執行; md 轉 html 為純 JS 不受此限
let pmChain = Promise.resolve()
let nQueue = 0


/**
 * 將函數排入Word轉檔佇列並逐一執行
 *
 * 底層係調用本機Word(win32com)轉檔，同時執行多份會相互干擾，故docx階段一律排隊逐一執行；前一件不論成敗均續行下一件。
 *
 * @param {Function} fun 輸入欲執行之函數，可回傳Promise
 * @returns {Promise} 回傳Promise，resolve回傳fun之結果，reject回傳錯誤
 * @example
 *
 * let r = await runExclusive(() => WHtml2docx('./a.html', './a.docx'))
 *
 */
function runExclusive(fun) {

    //check
    if (!isfun(fun)) {
        return Promise.reject('fun must be a function')
    }

    nQueue++
    let pm = pmChain.then(fun, fun) //前一件不論成敗均續行
    pmChain = pm.then(() => {
        nQueue--
    }, () => {
        nQueue--
    })
    return pm
}

 * @returns {Integer} 回傳排隊數整數
 * @example
 *
 * console.log(getQueueSize())
 * // => 0
 *
 */
function getQueueSize() {
    return nQueue
}


//fnExe: w-html2docx 之轉檔器檔名(其 bin/manifest.json 之 name)
let fnExe = 'htmlToDocx.exe'

//maxBytesRead: fs.readFileSync 讀入單一檔案之上限(2 GiB - 1 位元組), 超過即 ERR_FS_FILE_TOO_LARGE
let maxBytesRead = 2 ** 31 - 1

//maxCharsB64: 回傳內容以 base64 置於單一 JSON 字串, 該字串受 V8 字串長度上限(constants.MAX_STRING_LENGTH)限制,
//  另預留 1 MiB 給 JSON 外殼與其他欄位(名稱、檔名、keepCaption 與 toc 之結果等)
let maxCharsB64 = constants.MAX_STRING_LENGTH - 1024 * 1024


/**
 * 檢查本機是否具備docx轉檔條件
 *
 * ready僅以作業系統為Windows判定，不以轉檔器是否存在為條件：exeFound為false代表轉檔器目前尚未備妥(不在w-html2docx之快取內)，將於首次轉檔時由w-html2docx取得，並非不可用。
 *
 * exeFound與exePath以w-html2docx之ensureBins({offline:true,mirror:''})判斷：只看快取、不連網、不用鏡像，快取命中者亦校驗SHA-256與大小，故不會下載；校驗不符之快取檔會被其刪除。
 *
 * 注意：不得作為轉檔前之守門。查無不代表無法轉檔——w-html2docx於轉檔當下會取得缺少之轉檔器，上層以此攔截會使該機制永遠不被執行；亦僅代表環境條件成立，不代表轉檔器可取得或Word確實可被調用，後二者須實際轉一份小檔驗證。
 *
 * @returns {Promise} 回傳Promise，resolve回傳狀態物件{platform,isWindows,cwd,exePath,exeFound,ready}，exePath與exeFound為轉檔器在快取內之位置與是否已備妥(診斷用)
 * @example
 *
 * console.log(await checkDocxReady())
 * // => { platform: 'win32', isWindows: true, cwd: '...', exePath: 'C:\\proj\\node_modules\\w-html2docx\\bin-cache\\b9d244e2…\\htmlToDocx.exe', exeFound: true, ready: true }
 *
 */
async function checkDocxReady() {
    let exePath = ''
    await WHtml2docx.ensureBins({ offline: true, mirror: '' })
        .then((r) => {
            if (isestr(get(r, [fnExe]))) {
                exePath = r[fnExe]
            }
        })
        .catch(() => {
            //未備妥(E_OFFLINE_MISSING)或清單、校驗等其他失敗, 診斷上一律視為未備妥
        })
    let isWindows = process.platform === 'win32'
    return {
        platform: process.platform,
        isWindows,
        cwd: path.resolve(),
        exePath,
        exeFound: exePath !== '',
        ready: isWindows,
    }
}


/**
 * 確認單一檔案可由fs.readFileSync一次讀入
 *
 * fs.readFileSync讀入單一檔案有2 GiB-1位元組之上限，超過即拋出ERR_FS_FILE_TOO_LARGE(訊息未指出是哪個檔)；此處於讀檔前以檔案大小檢查，超過時拋出指出原因之錯誤。
 *
 * @param {String} fp 輸入檔案位置字串
 * @param {String} what 輸入檔案種類說明字串，用於錯誤訊息開頭，如'docx'
 * @example
 *
 * assertReadable('./report.docx', 'docx')
 * // => 超過2 GiB-1位元組時拋出Error: docx is too large to be read into memory in one piece (2147483648 bytes, over the limit of 2147483647 bytes = 2 GiB)
 *
 */
function assertReadable(fp, what) {
    let size = fs.statSync(fp).size
    if (size > maxBytesRead) {
        throw new Error(`${what} is too large to be read into memory in one piece (${size} bytes, over the limit of ${maxBytesRead} bytes = 2 GiB)`)
    }
}


/**
 * 確認各檔轉為base64後可合併置於單一JSON字串回傳
 *
 * 回傳內容(docx、html)以base64置於JSON字串，字串長度有上限(V8，並預留1 MiB給JSON外殼)；各檔base64字元數(每3位元組4字元，含進位)合計超過時拋出指出原因之錯誤，使呼叫端於讀檔與轉base64前即得到明確訊息，而非ERR_STRING_TOO_LONG。
 *
 * @param {Object} sizes 輸入各檔大小物件，格式{名稱:位元組數}，如{docx:409114973,html:3200}
 * @example
 *
 * assertB64Fit({ docx: 409114973 })
 * // => 拋出Error: the result is too large to be returned as base64 in one JSON string: docx 409114973 bytes (545486632 chars of base64 in total), over the limit of 535822312 chars (about 401866734 bytes of files in total)
 *
 */
function assertB64Fit(sizes) {
    let ks = Object.keys(sizes)
    let nChars = 0
    for (let k of ks) {
        nChars += 4 * Math.ceil(sizes[k] / 3)
    }
    if (nChars > maxCharsB64) {
        let cSizes = ks.map((k) => `${k} ${sizes[k]} bytes`).join(', ')
        throw new Error(`the result is too large to be returned as base64 in one JSON string: ${cSizes} (${nChars} chars of base64 in total), over the limit of ${maxCharsB64} chars (about ${Math.floor(maxCharsB64 / 4) * 3} bytes of files in total)`)
    }
}


/**
 * 取安全檔名主體
 *
 * 移除路徑成分、常見副檔名與跨平台非法字元，避免路徑穿越與寫檔失敗；無效輸入回傳'output'。
 *
 * @param {String} name 輸入檔名字串
 * @returns {String} 回傳安全檔名主體字串
 * @example
 *
 * console.log(toSafeName('../report/報告R00.01.md'))
 * // => '報告R00.01'
 *
 * console.log(toSafeName(''))
 * // => 'output'
 *
 */
function toSafeName(name) {
    let s = ''
    if (isstr(name)) {
        s = name.trim()
    }
    if (s === '') {
        return 'output'
    }
    s = path.basename(s) //去除任何路徑成分
    s = s.replace(/\.(md|markdown|html|htm|docx)$/i, '') //去除常見副檔名
    s = s.replace(/[\\/:*?"<>|\r\n\t]/g, '_') //Windows 非法字元
    s = s.replace(/^\.+/, '') //去除開頭點號(避免隱藏檔與相對路徑)
    s = s.trim()
    if (s === '') {
        return 'output'
    }
    return s.slice(0, 120)
}


//checkAssetPath: 單一相對路徑之落點檢查
//why: 路徑可能來自外部輸入, 須以絕對路徑比對且尾端補分隔字元, 避免同前綴之兄弟資料夾被放行
function checkAssetPath(dirJob, rp, pOri) {

    //拒絕絕對路徑(含磁碟機代號與 UNC)
    if (/^[a-zA-Z]:/.test(rp) || rp.startsWith('/') || rp.startsWith('//')) {
        throw new Error(`asset.path must be a relative path: ${pOri}`)
    }

    //落點檢查
    let fp = path.resolve(dirJob, rp)
    let base = path.resolve(dirJob) + path.sep
    if (!fp.startsWith(base)) {
        throw new Error(`asset.path is outside the working folder: ${pOri}`)
    }

    return fp
}


/**
 * 解析資產之相對路徑並防護路徑穿越
 *
 * md內之圖片路徑可能為URL編碼(中文或空白)，而轉檔器係以md內之原字串組路徑取圖；故原字串與解碼後字串不同時回傳兩者，供呼叫端寫入同一內容確保圖片必能被解析。
 *
 * @param {String} dirJob 輸入工作資料夾位置字串
 * @param {String} p 輸入資產相對路徑字串
 * @returns {Array} 回傳實際寫入之絕對路徑陣列
 * @example
 *
 * console.log(resolveAssetPaths('C:\\job', 'pics/%E5%9C%96.png'))
 * // => [ 'C:\\job\\pics\\%E5%9C%96.png', 'C:\\job\\pics\\圖.png' ]
 *
 */
function resolveAssetPaths(dirJob, p) {

    //raw
    let raw = ''
    if (isstr(p)) {
        raw = p.trim().replace(/\\/g, '/')
    }
    if (raw === '') {
        throw new Error('asset.path is empty')
    }

    //解碼; 非法 percent 序列則視為未編碼
    let dec = raw
    try {
        dec = decodeURIComponent(raw)
    }
    catch (err) {
        dec = raw
    }

    let rps = (raw === dec) ? [raw] : [raw, dec]
    return rps.map((rp) => checkAssetPath(dirJob, rp, p))
}


/**
 * 將夾帶之資產(圖片等)還原至工作資料夾
 *
 * 使md內之相對路徑得以解析；base64容許data URI形式。
 *
 * @param {String} dirJob 輸入工作資料夾位置字串
 * @param {Array} assets 輸入資產陣列，格式[{path,base64}]，亦接受contentBase64欄位
 * @returns {Integer} 回傳寫入之資產數整數
 * @example
 *
 * let n = writeAssets('C:\\job', [{ path: 'pics/圖.png', base64: '…' }])
 * // => 1
 *
 */
function writeAssets(dirJob, assets) {
    let n = 0
    if (!isarr(assets)) {
        return n
    }
    for (let v of assets) {
        let p = get(v, 'path', '')
        let fps = resolveAssetPaths(dirJob, p)
        let b64 = get(v, 'base64', '')
        if (!isestr(b64)) {
            b64 = get(v, 'contentBase64', '')
        }
        if (!isestr(b64)) {
            throw new Error(`asset.base64 is empty: ${p}`)
        }
        b64 = b64.replace(/^data:[^;]+;base64,/, '') //容許 data URI 形式
        let buf = Buffer.from(b64, 'base64')
        for (let fp of fps) {
            fs.mkdirSync(path.dirname(fp), { recursive: true })
            fs.writeFileSync(fp, buf)
        }
        n++
    }
    return n
}


/**
 * 清除工作資料夾內逾時未刪之作業夾(前綴job_)
 *
 * 工作資料夾不存在時自動建立。
 *
 * @param {String} dirWork 輸入工作資料夾位置字串
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Number} [opt.msAge=21600000] 輸入視為逾時之毫秒數，預設6小時(21600000)
 * @returns {Integer} 回傳刪除之作業夾數整數
 * @example
 *
 * let n = cleanWorkDir('./tmp', { msAge: 3600 * 1000 })
 * // => 2
 *
 */
function cleanWorkDir(dirWork, opt = {}) {

    //check
    if (!isestr(dirWork)) {
        return 0
    }

    //msAge
    let msAge = get(opt, 'msAge', null)
    if (!ispnum(msAge)) {
        msAge = 6 * 3600 * 1000
    }
    msAge = cdbl(msAge)

    if (!fsIsFolder(dirWork)) {
        fs.mkdirSync(dirWork, { recursive: true })
        return 0
    }
    let n = 0
    let msNow = Date.now()
    for (let fn of fs.readdirSync(dirWork)) {
        if (!fn.startsWith('job_')) {
            continue
        }
        let fp = path.resolve(dirWork, fn)
        try {
            let st = fs.statSync(fp)
            if (msNow - st.mtimeMs > msAge) {
                fs.rmSync(fp, { recursive: true, force: true })
                n++
            }
        }
        catch (err) {
            //忽略無法處理者
        }
    }
    return n
}


/**
 * 轉譯資產缺漏之錯誤訊息
 *
 * w-md2html於md引用之圖片不存在時reject「N image file(s) do not exist:」後逐行列出缺圖之絕對路徑(含內部工作夾路徑)，此處改為可行動之提示並隱去工作路徑，缺1個或多個各依單複數；缺圖路徑不全在工作夾內(如md內寫絕對路徑)或非資產缺漏訊息則原樣回傳。w-html2docx於Word解析不到圖時每行列出之「unresolved linked picture: <來源>」，來源為工作夾內之本機路徑者亦隱去工作路徑，網址與工作夾外之路徑原樣。
 *
 * @param {String} msg 輸入原始錯誤訊息字串
 * @returns {String} 回傳轉譯後錯誤訊息字串
 * @example
 *
 * console.log(transErrAsset('1 image file(s) do not exist:\nC:/tmp/job_1/pics/a.png'))
 * // => 'Failed to convert md to html: the referenced file (pics/a.png) does not exist, please make sure it is included in assets'
 *
 */
function transErrAsset(msg) {
    if (!isstr(msg)) {
        return toErrText(msg)
    }
    let mt = msg.match(/\d+ image file\(s\) do not exist:\r?\n([\s\S]*)$/)
    if (mt) {
        let rps = []
        for (let line of mt[1].split(/\r?\n/)) {
            if (line.trim() === '') {
                continue
            }
            let m = line.match(/job_[^\\/]+[\\/](.+)$/)
            if (!m) {
                return msg
            }
            rps.push(m[1].replace(/\\/g, '/'))
        }
        if (rps.length === 1) {
            return `Failed to convert md to html: the referenced file (${rps[0]}) does not exist, please make sure it is included in assets`
        }
        if (rps.length > 1) {
            return `Failed to convert md to html: the referenced files (${rps.join(', ')}) do not exist, please make sure they are included in assets`
        }
    }
    return msg.replace(/(unresolved linked picture: )[A-Za-z]:[\\/][^\r\n]*?job_[^\\/\r\n]+[\\/]([^\r\n]+)/g, (all, pre, rp) => `${pre}${rp.replace(/\\/g, '/')}`)
}


/**
 * 整理服務端請求之開關或設定物件
 *
 * 值為true或字串'true'時回傳true；為物件時回傳移除omit所列鍵後之複本；其餘回傳false。供toc、keepCaption等「true或設定物件」型之請求參數共用。
 *
 * @param {*} v 輸入請求之值
 * @param {Array} [omit=[]] 輸入不由請求端決定而須移除之鍵陣列，預設[]
 * @returns {Boolean|Object} 回傳false、true或設定物件
 * @example
 *
 * console.log(normReqOpt('true'))
 * // => true
 *
 * console.log(normReqOpt({ a: 1, b: 2 }, ['b']))
 * // => { a: 1 }
 *
 */
function normReqOpt(v, omit = []) {
    if (v === true || v === 'true') {
        return true
    }
    if (isobj(v)) {
        let o = { ...v }
        for (let k of isarr(omit) ? omit : []) {
            delete o[k]
        }
        return o
    }
    return false
}


/**
 * 整理服務端請求之目錄設定
 *
 * 同normReqOpt，另移除timeoutMs：Word更新之逾時屬服務端資源控管，不由請求端決定，否則請求端可用極短逾時反覆中斷Word更新，使經COM啟動之WINWORD行程殘留。
 *
 * @param {*} toc 輸入請求之toc
 * @returns {Boolean|Object} 回傳false、true或設定物件
 * @example
 *
 * console.log(normTocReq({ maxLevels: 2, timeoutMs: 1 }))
 * // => { maxLevels: 2 }
 *
 */
function normTocReq(toc) {
    return normReqOpt(toc, ['timeoutMs'])
}


/**
