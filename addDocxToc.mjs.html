import fs from 'fs'
import os from 'os'
import path from 'path'
import get from 'lodash-es/get.js'
import isestr from 'wsemi/src/isestr.mjs'
import isobj from 'wsemi/src/isobj.mjs'
import ispint from 'wsemi/src/ispint.mjs'
import cint from 'wsemi/src/cint.mjs'
import fsIsFile from 'wsemi/src/fsIsFile.mjs'
import execProcess from 'wsemi/src/execProcess.mjs'
import { prepDocxToc, checkDocxToc } from './docxToc.mjs'
import { toErrText, retryBusy, runExclusive, assertReadable } from './utils.mjs'


//ENV_SRC, ENV_DST: 腳本讀取輸入、輸出 docx 絕對路徑之環境變數名稱(路徑不拼進腳本字串, 免引號與注入問題)
let ENV_SRC = 'W_MD2DOCX_SRC'
let ENV_DST = 'W_MD2DOCX_DST'


//psScript: 以本機 Word 更新欄位並另存之 Windows PowerShell 5.1 腳本(僅含 ASCII)
//  why: 以 powershell.exe(Windows 內建)經 COM 驅動 Word, 腳本以 -EncodedCommand 帶於命令列, 不寫成檔案; 獨立之 .vbs 曾遭防毒軟體誤判隔離(Gitea #53), VBScript 亦將自 Windows 移除
//  輸出格式、離開碼與舊 vbs 相同: 成功印 `toc=<n> tof=<n> pages=<n>` 與 `ok` 後離開碼 0; 失敗印 `error: <原因>` 後離開碼 1(字頭依階段: cannot start Microsoft Word / open failed / update failed / save failed); 缺環境變數為用法錯誤, 離開碼 2
//  各階段以 try/catch 分段(對應舊 vbs 之 On Error Resume Next 與分段檢查), 失敗時先印錯誤再關閉文件與 Word, 與舊 vbs 之順序相同; 成功時 `ok` 印於關閉 Word 之後
//  Close、Quit 之 ByRef 參數須以 [ref] 傳(實測 Word 16: `$word.Quit(0)` 擲「引數應該是 PSReference」, 被收尾之 catch 吞掉, WINWORD 即殘留不結束); 收尾之 catch 不輸出, 以測試檢查無殘留
//  $ProgressPreference 置於首行: 否則 -EncodedCommand 在 stderr 被導向時會先以 CLIXML 輸出「正在準備模組」之進度; stdout 設為 UTF-8, 使 Word 回報之中文原因不受系統字碼頁影響
let psScript = [
    `$ProgressPreference = 'SilentlyContinue'`,
    `$ErrorActionPreference = 'Stop'`,
    `[Console]::OutputEncoding = New-Object System.Text.UTF8Encoding($false)`,
    `function Say($s) { [Console]::Out.WriteLine($s) }`,
    `function Why($e) {`,
    `    $x = $e.Exception`,
    `    while ($x.InnerException) { $x = $x.InnerException }`,
    `    ($x.Message -replace '\\s+', ' ').Trim()`,
    `}`,
    `$src = $env:${ENV_SRC}`,
    `$dst = $env:${ENV_DST}`,
    `if (-not $src -or -not $dst) {`,
    `    Say 'error: usage: set the environment variables ${ENV_SRC} and ${ENV_DST} (absolute paths of the source and the destination docx)'`,
    `    exit 2`,
    `}`,
    `$word = $null`,
    `$doc = $null`,
    `$code = 0`,
    `do {`,
    `    try { $word = New-Object -ComObject Word.Application }`,
    `    catch { Say ('error: cannot start Microsoft Word: ' + (Why $_)); $code = 1; break }`,
    `    try {`,
    `        $word.Visible = $false`,
    `        $word.DisplayAlerts = 0`,
    `        $doc = $word.Documents.Open($src, $false, $false, $false)`,
    `    }`,
    `    catch { Say ('error: open failed: ' + (Why $_)); $code = 1; break }`,
    `    try {`,
    `        $null = $doc.Fields.Update()`,
    `        $doc.Repaginate()`,
    `        foreach ($t in $doc.TablesOfContents) { $t.Update() }`,
    `        foreach ($f in $doc.TablesOfFigures) { $f.Update() }`,
    `        $doc.Repaginate()`,
    `        foreach ($t in $doc.TablesOfContents) { $t.UpdatePageNumbers() }`,
    `        foreach ($f in $doc.TablesOfFigures) { $f.UpdatePageNumbers() }`,
    `    }`,
    `    catch { Say ('error: update failed: ' + (Why $_)); $code = 1; break }`,
    `    try {`,
    `        Say ('toc=' + $doc.TablesOfContents.Count + ' tof=' + $doc.TablesOfFigures.Count + ' pages=' + $doc.ComputeStatistics(2))`,
    `        $doc.SaveAs2($dst, 12)`,
    `    }`,
    `    catch { Say ('error: save failed: ' + (Why $_)); $code = 1; break }`,
    `} while ($false)`,
    `if ($doc) { try { $doc.Close([ref]0) } catch {} }`,
    `if ($word) { try { $word.Quit([ref]0) } catch {} }`,
    `foreach ($o in @($doc, $word)) { if ($o) { try { [void][System.Runtime.InteropServices.Marshal]::ReleaseComObject($o) } catch {} } }`,
    `[GC]::Collect()`,
    `[GC]::WaitForPendingFinalizers()`,
    `if ($code -eq 0) { Say 'ok' }`,
    `exit $code`,
].join('\n')


//fromClixml: powershell.exe 在 -EncodedCommand 且 stderr 被導向時, 以 CLIXML(`#< CLIXML` 與 <Objs>)輸出進度與未捕捉之錯誤; 還原為純文字: 丟棄進度, 錯誤取其文字(`_xHHHH_` 為字元跳脫, 如換行為 _x000D__x000A_)
function fromClixml(s) {
    if (!/<Objs\b/.test(s)) {
        return s
    }
    let dec = (v) => v.replace(/&lt;/g, '<').replace(/&gt;/g, '>').replace(/&quot;/g, '"').replace(/&apos;/g, '\'').replace(/&amp;/g, '&').replace(/_x([0-9A-Fa-f]{4})_/g, (m, h) => String.fromCharCode(parseInt(h, 16)))
    return s
        .replace(/#< CLIXML\r?\n?/g, '')
        .replace(/<Objs\b[\s\S]*?<\/Objs>/g, (blk) => [...blk.matchAll(/<S S="Error">([\s\S]*?)<\/S>/g)].map((m) => dec(m[1])).join(''))
}


//execPs: 以 powershell.exe 執行腳本(-EncodedCommand, 不寫成檔案), env 為帶入腳本之環境變數物件
//  -NoProfile: 不載入使用者設定檔; -NonInteractive: 不出現提示; 不加 -ExecutionPolicy(執行原則只管腳本檔, 不影響 -EncodedCommand), 亦不加 -WindowStyle(execProcess 已隱藏視窗)
//  execProcess 無 env 選項, 故於執行期間暫設 process.env 並於落定後還原(子程序之啟動在 codeCmd 'auto' 時要等系統字碼頁查詢完, 不可於呼叫返回即還原); 呼叫端須序列化(addDocxToc 經 runExclusive)
//  codeCmd 'auto': stdout 為腳本設定之 UTF-8, stderr 為系統字碼頁(中文 Windows 為 cp950), 兩串流各自判定
//  resolve 離開碼 0 時之輸出字串; reject 為 execProcess 之錯誤字串(離開碼非 0 為 `code[N]:\n輸出`, 逾時為 `timeout[Nms]:\n輸出`), 皆已還原 CLIXML
async function execPs(script, env, timeout) {
    let keys = Object.keys(env)
    let olds = keys.map((k) => process.env[k])
    keys.forEach((k) => {
        process.env[k] = env[k]
    })
    let args = ['-NoProfile', '-NonInteractive', '-EncodedCommand', Buffer.from(script, 'utf16le').toString('base64')]
    try {
        let r = await execProcess('powershell.exe', args, { codeCmd: 'auto', timeout })
        return fromClixml(String(r))
    }
    catch (e) {
        return Promise.reject(fromClixml(toErrText(e)))
    }
    finally {
        keys.forEach((k, i) => {
            if (olds[i] === undefined) {
                delete process.env[k]
            }
            else {
                process.env[k] = olds[i]
            }
        })
    }
}


//runWord: 以本機 Word 更新欄位並另存(PowerShell 經 COM 驅動, 見 psScript)
//  逾時時 execProcess 結束 powershell.exe 之程序樹, 惟經 COM 啟動之 WINWORD 不在其中, 可能殘留
async function runWord(fpIn, fpOut, timeout) {
    let out = ''
    let err = null
    await execPs(psScript, { [ENV_SRC]: fpIn, [ENV_DST]: fpOut }, timeout)
        .then((r) => {
            out = String(r).trim()
        })
        .catch((e) => {
            err = toErrText(e).trim()
        })
    if (err !== null) {
        return Promise.reject(`Word failed to update the fields: ${err}`)
    }
    if (!/(^|\n)ok$/.test(out)) {
        return Promise.reject(`Word failed to update the fields: ${out}`)
    }
    let m = out.match(/pages=(\d+)/)
    return { pages: m ? Number(m[1]) : null }
}


/**
 * 於docx添加章節目錄、圖目錄、表目錄並重編頁碼
 *
 * 先以純結構處理準備內容(見docxToc.mjs之prepDocxToc：圖名表名編號改SEQ欄位、標題粗體改由樣式提供、插入目錄欄位、分節與頁碼)，再以本機Microsoft Word更新欄位(目錄內容與頁碼只有Word能算)，最後逐項核對目錄項目、連結落點、頁碼與分節，全部相符才覆寫原檔。故僅能於已安裝Word之Windows執行，Word之調用與轉檔共用同一佇列。
 *
 * 頁碼：第1個標題前有內容者視為封面，封面無頁碼；前置章節與目錄為大寫羅馬數字(I起)；正文自1起。無封面者不另建封面頁。
 *
 * 標題以段落之大綱階層判斷，第1個標題即視為正文之始(前置章節除外)：Markdown以單一#為文件標題、##為章者，文件標題會列為目錄第1層且正文自其起算；欲置於封面請以非標題之段落撰寫(如<div pretitle>)。
 *
 * Word更新以Windows內建之PowerShell 5.1(powershell.exe)經COM驅動，腳本以-EncodedCommand帶於命令列、不寫成檔案(獨立之腳本檔會被防毒軟體誤判隔離，且VBScript已公告淘汰，故不使用)；故需Windows、Microsoft Word與powershell.exe。逾時被終止時，經COM啟動之WINWORD行程可能殘留，須人工確認後結束。
 *
 * 文件無標題段落時不處理並於回傳之skip說明原因；前提不符或核對不符時reject，原檔不變。docx須整份讀入記憶體，單檔上限為2 GiB-1位元組(fs.readFileSync)，超過時於讀檔前以指出原因之訊息reject，原檔不變。
 *
 * @param {String} fpDocx 輸入docx檔位置字串，處理成功時就地覆寫
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {Object} [opt.labels={fig:'圖',tab:'表'}] 輸入圖名、表名之標籤(段首「圖N」「表N」，編號後須接空白)，須為不含空白與引號之非空字串且兩者不同，英文報告可給{fig:'Figure',tab:'Table'}
 * @param {Object} [opt.titles={toc:'目錄',fig:'圖目錄',tab:'表目錄'}] 輸入三目錄之標題
 * @param {Array} [opt.frontHeadings=['摘要','ABSTRACT','Abstract']] 輸入前置章節之標題文字，文件開頭之標題與之相同者排在封面與目錄之間且不列入目錄
 * @param {Integer} [opt.maxLevels=3] 輸入目錄列出之標題層數(自正文最高之大綱階層起算)，上限9
 * @param {String} [opt.titleStyle='報告目錄標題'] 輸入三目錄標題之段落樣式名稱(預設為本套件所附temp.docx之樣式)，模板無此樣式時另建置中粗體樣式；此名稱會出現在產出docx之樣式清單，交付給業主時須為中性名稱；繁體中文版Word視「目錄標題」為內建樣式TOC Heading之本地化名稱，自訂樣式同名時會被更名為「目錄標題1」，故勿用該名稱
 * @param {Array} [opt.headStyles=['章標題','節標題','小節標題']] 輸入承接標題粗體之段落樣式名稱(依層)，模板已有同名樣式時須為粗體；此名稱亦會出現在產出docx之樣式清單，交付給業主時須為中性名稱
 * @param {Boolean} [opt.pageNumbers=true] 輸入是否分節重編頁碼，false時只插入三目錄(各自起新頁)，不動分節與頁碼
 * @param {Boolean} [opt.keepLines=true] 輸入是否將目錄來源段落(列入目錄之標題、圖名、表名)設為段落內不分頁，使目錄之頁碼即該段落所在之頁(多行之表名跨頁時，首行留在前頁而表格在次頁，表目錄之頁碼會指向前頁)
 * @param {Boolean} [opt.keepWithObject=true] 輸入是否使圖名、表名與其圖、表同頁(與keepCaption同一規則，見docxKeep.mjs之planKeepNext)：與下段之表格或圖片、或上段之圖片同頁，上下皆有圖表時依同類圖名表名之多數位置，平手時圖名取上段、表名取下段(圖在頁底而圖名被擠到次頁時，圖目錄之頁碼會指向次頁)
 * @param {Boolean} [opt.replaceTocStyles=true] 輸入是否以本套件之目錄項目樣式(第1層粗體、下層縮排並懸掛縮排、點線引導頁碼，字型沿用正文第1個標題)取代模板既有之toc N與table of figures樣式，false時沿用模板既有者、僅補建缺少者
 * @param {Integer} [opt.timeoutMs=1800000] 輸入Word更新之逾時毫秒數，預設1800000(30分鐘)
 * @returns {Promise} 回傳Promise，resolve回傳{skip,cover,levels,front,lists,toc,fig,tab,pagesFirst,pages,warns,ms}(toc、fig、tab為各目錄之項目數，lists為實際插入之目錄，pagesFirst為目錄第1項之頁碼)，無標題時僅回傳{skip,ms}；reject回傳錯誤訊息字串
 * @example
 *
 * import addDocxToc from 'w-md2docx/src/addDocxToc.mjs'
 *
 * let r = await addDocxToc('./report.docx')
 * console.log(r)
 * // => { skip: '', cover: true, levels: [1, 2], front: [], lists: ['toc', 'fig', 'tab'], toc: 58, fig: 246, tab: 55, pagesFirst: '1', pages: 170, warns: [], ms: 12500 }
 *
 */
async function addDocxToc(fpDocx, opt = {}) {
    let msStart = Date.now()

    //check
    if (process.platform !== 'win32') {
        return Promise.reject('addDocxToc requires Microsoft Word on Windows')
    }
    if (!isestr(fpDocx)) {
        return Promise.reject('fpDocx must be a non-empty string')
    }
    fpDocx = path.resolve(fpDocx)
    if (!fsIsFile(fpDocx)) {
        return Promise.reject(`fpDocx[${fpDocx}] does not exist`)
    }

    //opt
    if (!isobj(opt)) {
        opt = {}
    }

    //timeoutMs
    let timeoutMs = get(opt, 'timeoutMs', null)
    if (!ispint(timeoutMs)) {
        timeoutMs = 1800000
    }
    timeoutMs = cint(timeoutMs)

    //準備內容(前提不符即拋錯, 原檔不變)
    let prep = null
    try {
        assertReadable(fpDocx, 'docx') //整份讀入記憶體, 超過 fs.readFileSync 上限時於讀檔前以明確訊息拋出
        prep = prepDocxToc(new Uint8Array(fs.readFileSync(fpDocx)), opt)
    }
    catch (err) {
        return Promise.reject(`Failed to prepare the TOC: ${toErrText(err)}`)
    }
    let info = prep.info
    if (info.skip) {
        return { skip: info.skip, ms: Date.now() - msStart }
    }

    //Word 更新欄位(與轉檔共用佇列), 核對全部相符才覆寫原檔
    let tag = `${process.pid}_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`
    let fpTmpIn = path.join(os.tmpdir(), `wmd2docx_toc_${tag}_in.docx`)
    let fpTmpOut = path.join(os.tmpdir(), `wmd2docx_toc_${tag}_out.docx`)
    try {
        fs.writeFileSync(fpTmpIn, prep.u8)
        let rWord = await runExclusive(() => runWord(fpTmpIn, fpTmpOut, timeoutMs))
        if (!fsIsFile(fpTmpOut) || fs.statSync(fpTmpOut).size === 0) {
            return Promise.reject('Word reported success but the updated docx was not generated')
        }
        assertReadable(fpTmpOut, 'docx updated by Word')
        let chk = checkDocxToc(new Uint8Array(fs.readFileSync(fpTmpOut)), info)
        if (!chk.ok) {
            return Promise.reject(`TOC check failed (${chk.errs.length}): ${chk.errs.slice(0, 5).join('; ')}`)
        }
        await retryBusy(() => fs.promises.copyFile(fpTmpOut, fpDocx))
        return {
            skip: '',
            cover: info.cover,
            levels: info.levels,
            front: info.front,
            lists: info.lists,
            toc: chk.stats.toc,
            fig: chk.stats.fig,
            tab: chk.stats.tab,
            pagesFirst: chk.stats.pagesFirst,
            pages: rWord.pages,
            warns: info.warns,
            ms: Date.now() - msStart,
        }
    }
    catch (err) {
        return Promise.reject(toErrText(err))
    }
    finally {
        for (let fp of [fpTmpIn, fpTmpOut]) {
            try {
                if (fsIsFile(fp)) {
                    fs.unlinkSync(fp)
                }
            }
            catch (err) {
                //暫存檔被鎖(如逾時殘留之 WINWORD 仍開啟), 留待系統暫存清理
            }
        }
    }
}
